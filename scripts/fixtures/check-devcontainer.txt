# Cases of scripts/check-devcontainer.sh, read by scripts/test-check-devcontainer.sh.
#
# One case per line, four fields separated by ' | ':
#
#   <pass or fail> | <what the output holds, or -> | <name> | <tree>
#
# A run exits 0 for pass and 1 for fail. The second field is '<key>', then ' @ <subject>' for each
# datum the line names, such as the clause. A line of the output ends in '[<key>]', the rule and a
# reason word, and holds each subject, in that order and with no letter, digit, hyphen, or
# underscore beside it; the rest of its wording is free.
#
# The tree is a list of steps separated by '; ', applied in order to a repository that passes the
# check, not under git, of three files. .devcontainer/devcontainer.json:
#
#   {
#     "name": "fixture",
#     // A comment: the container has no socket mount.
#     "image": "mcr.microsoft.com/devcontainers/base:debian",
#     "postStartCommand": <post>,
#     "features": {
#       // The one Feature.
#       "ghcr.io/devcontainers/features/github-cli:1": {}
#     },
#     "customizations": { "vscode": { "extensions": [] } }
#   }
#
# where <post> is the one line
#
#   { "hooks": "git config core.hooksPath .githooks", "merge": "git config merge.ff false && git config pull.ff only", "index": "git config merge.adr-index.driver '.githooks/merge-adr-index %O %A %B %S %X %Y'" }
#
# .devcontainer/devcontainer-lock.json:
#
#   {
#     "features": {
#       "ghcr.io/devcontainers/features/github-cli:1": {
#         "version": "1.0.0",
#         "resolved": "example@sha256:0000",
#         "integrity": "sha256:0000"
#       },
#     }
#   }
#
# .vscode/settings.json:
#
#   {
#     // Keep the container-management extension on the host side.
#     "remote.extensionKind": { "ms-azuretools.vscode-containers": ["ui"] }
#   }
#
# The steps:
#
#   rm <path>                   the file or directory removed
#   extra <json>                devcontainer.json written again, the json as one more top-level
#                               line before the features block
#   post [json]                 devcontainer.json written again, the json as the value of
#                               postStartCommand, or without that line when no json is given
#   settings <json>             .vscode/settings.json written again, the json as the value of
#                               remote.extensionKind
