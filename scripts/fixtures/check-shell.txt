# Cases of the rules shell/root and shell/bash-floor of scripts/check-shell.sh, read by
# scripts/test-check-shell.sh. The cases of ShellCheck, sed, and grep stay inline there: their
# findings carry no key yet.
#
# One case per line, four fields separated by ' | ':
#
#   <pass or fail> | <what the output holds, or -> | <name> | <tree>
#
# A run exits 0 for pass and 1 for fail. The second field is '<key>', then ' @ <subject>' for each
# datum the line names, such as the script. A line of the output ends in '[<key>]', the rule and a
# reason word, and holds each subject, in that order and with no ASCII letter, digit, hyphen,
# or underscore beside it; the rest of its wording is free.
#
# A line that starts with '#' is a comment, not a case. The subjects are looked for in the part of
# the line before its key, so that none holds by the key alone. A field may start with '!': it then
# holds where no line that ends in '[<key>]' holds the subjects in their order.
#
# The tree is a list of steps separated by '; ', or '-' for none, applied in order to a git
# repository with no commit that holds a copy of this repository's scripts/lib/ and a
# .shellcheckrc that disables SC1091 and SC2154, since a fixture's script sources what ShellCheck
# does not follow. The steps:
#
#   script <path> <kind>   a bash script at the path, of one of these kinds:
#                          rooted    sources the bash floor and scripts/lib/root.sh, calls
#                                    enter_root on a line of its own and rest_args, and prints its
#                                    root
#                          unrooted  prints a word, and sources nothing
#                          noargs    as rooted, without rest_args: it takes any argument
#                          ownroot   as rooted, with an enter_root of its own that reads no root
#                          floor     sources the bash floor of scripts/lib/ and prints a word
#                          floorless sources lib/bash.sh beside it and prints a word
#   write <path> <text>    the file written with the text and a line break, its directories made

pass | - | a script that takes its root through root.sh | script scripts/build.sh rooted
fail | shell/root:unrooted @ scripts/build.sh @ scripts/lib/root.sh @ CONTRIBUTING.md @ NUC-checks-are-bash @ §9 | a project's script that calls no enter_root | script scripts/build.sh unrooted
fail | shell/root:verbose @ scripts/build.sh | a script without rest_args, which takes --verbose | script scripts/build.sh noargs
fail | shell/root:missing @ scripts/build.sh | a script whose enter_root is its own reads a missing root | script scripts/build.sh ownroot
pass | - | a self-test is not held to the root | script scripts/test-build.sh unrooted
pass | - | check-all.sh is not held to the root | script scripts/check-all.sh unrooted
pass | - | a script below scripts/ is not held | script scripts/tools/build.sh unrooted
pass | - | a script outside scripts/ is not held | script tools/build.sh unrooted
pass | - | a skill script is not held to the root | script .agents/skills/x/run.sh unrooted
pass | - | a skill script that sources the bash floor stops below bash 4.4 | script .agents/skills/x/run.sh floor
fail | shell/bash-floor:silent @ .agents/skills/x/run.sh | a skill script whose floor does not stop it | write .agents/skills/x/lib/bash.sh # shellcheck shell=bash; script .agents/skills/x/run.sh floorless
fail | shell/bash-floor:silent @ scripts/test-build.sh | a self-test whose floor does not stop it | write scripts/lib/bash.sh # shellcheck shell=bash; script scripts/test-build.sh floor
pass | - | a script below scripts/ is not held to the bash floor | write scripts/tools/lib/bash.sh # shellcheck shell=bash; script scripts/tools/build.sh floorless
